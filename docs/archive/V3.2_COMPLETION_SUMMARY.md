## 🚀 集成使用示例

### 完整生成流程

```python
# 1. 获取账号人设
persona = persona_manager.get_persona(account_id)

# 2. 获取激活的 Prompt 版本
prompt_version = version_manager.get_active_version(
    VersionType.PROMPT,
    'content_generator'
)

# 3. 应用人设到 Prompt
enhanced_prompt = persona_manager.apply_persona_to_prompt(
    persona,
    prompt_version.content['system_prompt']
)

# 4. 开始全链路追踪
trace_id = observability_service.start_trace('content_generation')

# 5. 生成内容
span_id = observability_service.start_span(trace_id, 'generation')
title, text = await generator.generate(enhanced_prompt, ...)
observability_service.end_span(span_id, status='success')

# 6. 策略约束验证
is_valid, violations = strategy_controller.validate_content(title, text)
if not is_valid:
    # 重新生成或修正
    pass

# 7. 计算基础分数
base_score = await scorer.score(title, text, ...)

# 8. 检测并惩罚劫持行为
penalties = defense_system.calculate_penalty(text, keywords, templates)
score_after_defense = base_score - penalties['total']

# 9. 人设一致性调整
final_score = persona_manager.adjust_score_by_persona(
    persona, title, text, score_after_defense
)

# 10. 记录版本使用
version_manager.record_usage(
    version_id=prompt_version.version_id,
    generation_id=generation_id,
    success=True,
    result_metrics={'viral_score': final_score}
)

# 11. 存储特征
features = feature_extractor.extract_all_features(...)
feature_store.compute_and_store_note_features(note_id, features)

# 12. 记录观测性指标
observability_service.record_metric(
    metric_name='viral_score',
    metric_value=final_score,
    metric_type=MetricType.GAUGE,
    tags={'generation_id': generation_id}
)
```
