**检测方法**:
```python
class RewardDefenseSystem:
    def detect_keyword_stuffing(self, text, keywords):
        """检测关键词堆砌"""
        density = keyword_count / total_words
        is_stuffing = density > threshold  # 默认 5%
        return is_stuffing, density

    def detect_ngram_repetition(self, text, n=3):
        """检测 n-gram 重复"""
        # 统计所有 3-gram
        # 找出最大重复次数
        is_repetitive = max_repeat > threshold  # 默认 3 次
        return is_repetitive, max_repeat

    def detect_template_overuse(self, text, templates):
        """检测模板过度使用"""
        # 计算与模板的相似度
        is_overused = max_similarity > threshold  # 默认 0.8
        return is_overused, max_similarity

    def calculate_penalty(self, text, keywords, templates):
        """计算惩罚项"""
        penalties = {
            'keyword_stuffing': density_penalty,
            'ngram_repetition': repeat_penalty,
            'template_overuse': template_penalty,
            'total': sum(all_penalties)
        }
        return penalties
```

**应用到评分**:
```python
# 原始分数
base_score = 0.8

# 计算惩罚
penalties = defense.calculate_penalty(text, keywords, templates)

# 最终分数
final_score = base_score - penalties['total']
```

**效果**:
- ✅ 防止关键词堆砌
- ✅ 防止过度重复
- ✅ 防止模板化

**漂移检测算法**:
```python
# PSI（Population Stability Index）简化版
drift_score = (
    0.4 * mean_drift +      # 均值漂移
    0.2 * std_drift +       # 标准差漂移
    0.2 * p50_drift +       # 中位数漂移
    0.2 * p90_drift         # 90分位数漂移
)

# 各项漂移计算
mean_drift = abs(current_mean - baseline_mean) / (baseline_std + 1e-6)
std_drift = abs(current_std - baseline_std) / (baseline_std + 1e-6)
p50_drift = abs(current_p50 - baseline_p50) / (baseline_std + 1e-6)
p90_drift = abs(current_p90 - baseline_p90) / (baseline_std + 1e-6)
```

**健康度评分**:
```python
# 基础分数（关键指标）
base_score = (
    0.4 * generation_success_rate +
    0.3 * publish_success_rate +
    0.3 * min(avg_viral_score, 1.0)
)

# 漂移惩罚
drift_penalty = sum(d.drift_score for d in drifts if d.is_drifted) * 0.1

# 告警惩罚
alert_penalty = len([a for a in alerts if a.severity in [ERROR, CRITICAL]]) * 0.05

# 最终分数
health_score = max(0.0, base_score - drift_penalty - alert_penalty)

# 健康状态
if health_score >= 0.9: status = 'excellent'
elif health_score >= 0.7: status = 'good'
elif health_score >= 0.5: status = 'fair'
elif health_score >= 0.3: status = 'poor'
else: status = 'critical'
```

@celery_app.task
async def check_version_performance():
    """检查版本性能"""
    manager = VersionManager(db)

    # 获取所有激活版本
    active_versions = db.query(Version).filter(
        Version.status == VersionStatus.ACTIVE
    ).all()

    for version in active_versions:
        await manager.check_and_rollback(version.version_id)

@celery_app.task
async def monitor_feature_drift():
    """监控特征漂移"""
    store = FeatureStore(db)
    drifts = await store.monitor_all_features()

    if drifts:
        # 发送告警
        logger.warning(f"发现 {len(drifts)} 个漂移特征")
