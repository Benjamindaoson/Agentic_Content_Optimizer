## 🚀 快速开始

### 1. 基础内容生成

```python
from app.agents.workflow.langgraph_workflow import ContentGenerationWorkflow

# 创建工作流
workflow = ContentGenerationWorkflow()

# 生成内容
result = await workflow.generate_content(
    topic="AI 写作工具",
    platform="xiaohongshu",
    max_iterations=3
)

print(f"状态: {result['status']}")
print(f"评分: {result['final_score']}")
```

### 2. A2A 协同生成

```python
from app.messaging.message_bus import MessageBus
from app.agents.collaborative_generation import CollaborativeContentGenerator

# 创建消息总线
message_bus = MessageBus()
await message_bus.start()

# 创建协同生成器
generator = CollaborativeContentGenerator(message_bus)

# 协同生成
result = await generator.generate_with_realtime_feedback(
    topic="AI 写作工具",
    platform="xiaohongshu",
    max_iterations=3,
    quality_threshold=8.0
)
```

### 3. MCP 多平台发布

```python
from app.mcp import MultiPlatformManager, Platform

# 创建多平台管理器
manager = MultiPlatformManager()

# 发布到多个平台
results = await manager.publish_to_multiple_platforms(
    content={
        "hook": "AI 写作工具推荐",
        "body": "这是一个很棒的 AI 写作工具",
        "cta": "关注我了解更多"
    },
    platforms=[Platform.XIAOHONGSHU, Platform.DOUYIN, Platform.WEIBO],
    account_ids={
        Platform.XIAOHONGSHU: "xhs_account",
        Platform.DOUYIN: "dy_account",
        Platform.WEIBO: "wb_account"
    }
)
```
