```python
class ConnectionPool:

        self.pool = asyncpg.create_pool(
            dsn="postgresql://...",
            min_size=10,      # 最小连接数
```

#### 方案 D: 读写分离

```python
class ReadWriteSplitter:
    """读写分离"""

    def __init__(self):
        self.master = Database("master")  # 写
        self.slaves = [
            Database("slave1"),  # 读
            Database("slave2"),  # 读
            Database("slave3")   # 读
        ]
        self.current_slave = 0

    async def read(self, sql: str, *args):
        """读操作 → 从库"""
        slave = self.slaves[self.current_slave]
        self.current_slave = (self.current_slave + 1) % len(self.slaves)
        return await slave.query(sql, *args)

    async def write(self, sql: str, *args):
        """写操作 → 主库"""
        return await self.master.execute(sql, *args)

# 性能对比
# 单库: 100ms (高负载)
# 读写分离: 30ms (负载分散)
# 提升：延迟 -70%
```

```javascript
// Cloudflare Worker
export default {
  async fetch(request) {
    // 在边缘节点执行
    const cache = caches.default;

    // 1. 尝试缓存
    let response = await cache.match(request);
    if (response) {
      return response;  // < 10ms (边缘缓存)
    }

    // 2. 回源
    response = await fetch(request);

    // 3. 缓存
    await cache.put(request, response.clone());

    return response;
  }
}

// 性能对比
// 中心化: 200ms (跨地域延迟)
// 边缘计算: 10ms (就近访问)
```
