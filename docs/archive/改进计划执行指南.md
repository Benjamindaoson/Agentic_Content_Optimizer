from app.agents.planning import Planner, Reflector

class WriterAgent(BaseAgent):
    def __init__(self, ...):
        super().__init__(config)
        self.planner = Planner()
        self.reflector = Reflector()
        self.max_retries = 3

    async def execute(self, input_data: Dict[str, Any]) -> AgentResponse:
        # Phase 1: Planning
        plan = await self.planner.create_plan(
            task=f"为'{input_data['topic']}'生成{input_data['platform']}内容",
            context=input_data,
            max_steps=5
        )

        # Phase 2: Execution with Reflection
        for attempt in range(self.max_retries):
            results = []
            for step in plan.steps:
                result = await self._execute_step(step, input_data)

                # Reflection
                reflection = await self.reflector.reflect(
                    goal=step.goal,
                    result=result,
                    success_criteria=step.success_criteria
                )

                if reflection.needs_retry and attempt < self.max_retries - 1:
                    # 使用改进建议重试
                    result = await self._retry_with_feedback(
                        step,
                        result,
                        reflection.improvement_suggestions
                    )

                results.append(result)

            # 整体评估
            final_reflection = await self.reflector.reflect(
                goal="生成高质量内容",
                result=results[-1],
                success_criteria="内容质量评分 >= 8.0"
            )

            if not final_reflection.needs_retry:
                break

        return AgentResponse(success=True, data=results[-1])
```

from langgraph.graph import StateGraph, END
from typing import TypedDict

class ContentGenerationState(TypedDict):
    topic: str
    platform: str
    trend_analysis: dict
    references: list
    generated_content: dict
    evaluation: dict
    iteration: int
    max_iterations: int

class ContentGenerationWorkflow:
    def __init__(self, trend_agent, writer_agent, critic_agent):
        self.trend_agent = trend_agent
        self.writer_agent = writer_agent
        self.critic_agent = critic_agent
        self.workflow = self._build_workflow()

    def _build_workflow(self) -> StateGraph:
        workflow = StateGraph(ContentGenerationState)

        # 添加节点
        workflow.add_node("trend_analysis", self._trend_node)
        workflow.add_node("content_generation", self._writer_node)
        workflow.add_node("content_evaluation", self._critic_node)
        workflow.add_node("refinement", self._refinement_node)

        # 添加边
        workflow.set_entry_point("trend_analysis")
        workflow.add_edge("trend_analysis", "content_generation")
        workflow.add_edge("content_generation", "content_evaluation")

        # 条件边
        workflow.add_conditional_edges(
            "content_evaluation",
            self._should_refine,
            {"refine": "refinement", "end": END}
        )
        workflow.add_edge("refinement", "content_generation")

        return workflow.compile()

    async def run(self, input_data: dict) -> dict:
        state = ContentGenerationState(
            topic=input_data["topic"],
            platform=input_data["platform"],
            iteration=0,
            max_iterations=3
        )
        result = await self.workflow.ainvoke(state)
        return result
```

class RAGEvaluator:
    def calculate_ndcg(self, retrieved_docs, relevance_scores, k=10) -> float:
        """计算 NDCG@k"""
        dcg = sum((2 ** rel - 1) / np.log2(i + 2)
                  for i, rel in enumerate(relevance_scores[:k]))
        ideal_scores = sorted(relevance_scores, reverse=True)
        idcg = sum((2 ** rel - 1) / np.log2(i + 2)
                   for i, rel in enumerate(ideal_scores[:k]))
        return dcg / idcg if idcg > 0 else 0.0

    def calculate_mrr(self, retrieved_docs, relevant_doc_ids) -> float:
        """计算 MRR"""
        for i, doc in enumerate(retrieved_docs):

class OnlineLearningLoop:
    def __init__(self, grpo_trainer, metrics_collector):
        self.trainer = grpo_trainer
        self.metrics_collector = metrics_collector
        self.update_interval = 1000

    async def collect_and_train(self):
        while True:
            # 1. 收集真实用户反馈
            experiences = await self.metrics_collector.collect_recent(
                limit=self.update_interval
            )

            # 2. 计算真实奖励
            for exp in experiences:
                exp.reward = self._calculate_real_reward(exp.metrics)

            # 3. 更新 GRPO 模型
            result = await self.trainer.train_on_experiences(experiences)

            # 4. A/B 测试验证
            if result.loss_improved:
                await self._run_ab_test(result.new_model)

            await asyncio.sleep(3600)  # 每小时更新一次

    def _calculate_real_reward(self, metrics: dict) -> float:
        return (
            metrics.get('like_rate', 0) * 0.3 +
            metrics.get('comment_rate', 0) * 0.3 +
            metrics.get('collect_rate', 0) * 0.2 +
            metrics.get('share_rate', 0) * 0.2
        )
```

import mlflow

class ExperimentTracker:
    def __init__(self):
        mlflow.set_tracking_uri("http://localhost:5000")

    async def log_generation(self, params, metrics, artifacts):
        with mlflow.start_run():
            mlflow.log_params(params)
            mlflow.log_metrics(metrics)
            for name, content in artifacts.items():
                mlflow.log_text(content, f"{name}.txt")

    async def log_rl_training(self, episode, reward, loss, model_path):
        with mlflow.start_run():
            mlflow.log_metric("reward", reward, step=episode)
            mlflow.log_metric("loss", loss, step=episode)
            mlflow.log_artifact(model_path)

class SyntheticDataGenerator:
    async def generate_training_pairs(self, topic, num_samples=100):
        samples = []
        for i in range(num_samples):
            # 1. 生成多个候选
            candidates = await self._generate_candidates(topic, num=5)

            # 2. LLM Judge 评分
            scores = await self._llm_judge(candidates)

            # 3. 选择 best/worst 作为 preferred/rejected
            best_idx = scores.index(max(scores))
            worst_idx = scores.index(min(scores))

            samples.append({
                "topic": topic,
                "preferred": candidates[best_idx],
                "rejected": candidates[worst_idx],
                "score_diff": scores[best_idx] - scores[worst_idx]
            })

        return samples
```
