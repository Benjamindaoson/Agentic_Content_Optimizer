## 🚀 Fast Path (快速路径)

### 场景：查询类操作
- 获取话题列表
- 获取每日计划
- 获取封面候选

### 策略：缓存优先

```python
class FastPath:
    """快速路径：< 2s"""

    def __init__(self):
        self.cache = RedisCache()
        self.ttl = {
            'topics': 3600,      # 话题缓存 1 小时
            'plans': 86400,      # 计划缓存 1 天
            'covers': 604800     # 封面缓存 7 天
        }

    async def get_trending_topics(self, source: str) -> List[Dict]:
        """获取热点话题（缓存优先）"""
        cache_key = f"topics:{source}:{date.today()}"

        # 1. 尝试从缓存获取
        cached = await self.cache.get(cache_key)
        if cached:
            return cached  # < 10ms

        # 2. 缓存未命中，从数据库获取
        topics = await self.db.get_topics(source)
        if topics:
            await self.cache.set(cache_key, topics, ttl=self.ttl['topics'])
            return topics  # < 100ms

        # 3. 数据库也没有，异步触发采集
        await self.trigger_async_fetch(source)
        return []  # 返回空，下次请求会有数据

    async def get_daily_plan(self, account_id: str, date: date) -> Dict:
        """获取每日计划（缓存优先）"""
        cache_key = f"plan:{account_id}:{date}"

        cached = await self.cache.get(cache_key)
        if cached:
            return cached  # < 10ms

        plan = await self.db.get_plan(account_id, date)
        if plan:
            await self.cache.set(cache_key, plan, ttl=self.ttl['plans'])
            return plan  # < 100ms

        return None
```

## ⚡ Normal Path (常规路径)

### 场景：生成类操作
- 创建每日计划
- 生成内容
- 适配平台

### 策略：并行化 + 预计算

```python
class NormalPath:
    """常规路径：< 10s"""

    async def create_daily_plan(self, account_id: str) -> Dict:
        """创建每日计划（并行化）"""
        start_time = time.time()

        # 1. 并行获取数据（< 1s）
        topics, persona, history = await asyncio.gather(
            self.get_cached_topics(),      # 从缓存获取
            self.get_persona(account_id),  # 从数据库获取
            self.get_history(account_id)   # 从数据库获取
        )

        # 2. 并行计算（< 2s）
        selected, schedules = await asyncio.gather(
            self.select_topics(topics, persona, history),
            self.calculate_schedules(topics, history)
        )

        # 3. 组装计划（< 0.1s）
        plan = {
            'plan_id': generate_id(),
            'account_id': account_id,
            'selected_topics': selected,
            'schedules': schedules,
            'created_at': datetime.now()
        }

        # 4. 异步保存（不等待）
        asyncio.create_task(self.save_plan(plan))

        elapsed = time.time() - start_time
        logger.info(f"创建计划耗时: {elapsed:.2f}s")

        return plan  # < 3s

    async def generate_content_with_cover(
        self,
        topic: str,
        persona: Dict
    ) -> Dict:
        """生成内容 + 封面（并行化）"""
        start_time = time.time()

        # 1. 并行生成内容和封面（< 8s）
        content, covers = await asyncio.gather(
            self.generate_content(topic, persona),  # LLM: 3-5s
            self.generate_covers(topic, count=3)    # DALL·E: 5-8s
        )

        # 2. 并行预测 CTR（< 1s）
        for cover in covers:
            cover.predicted_ctr = await self.predict_ctr(cover)

        elapsed = time.time() - start_time
        logger.info(f"生成内容+封面耗时: {elapsed:.2f}s")

        return {
            'content': content,
            'covers': covers
        }  # < 9s
```

```python
class SlowPath:
    """慢速路径：异步执行"""

    async def estimate_causal_effect(
        self,
        treatment: str,
        outcome: str
    ) -> str:
        """估计因果效应（异步）"""
        # 1. 创建异步任务
        task_id = generate_id()
        task = {
            'task_id': task_id,
            'type': 'causal_inference',
            'params': {'treatment': treatment, 'outcome': outcome},
            'status': 'pending',
            'created_at': datetime.now()
        }

        await self.db.save_task(task)

        # 2. 提交到 Celery
        celery_task = estimate_causal_effect_task.delay(
            treatment, outcome
        )

        # 3. 立即返回任务 ID
        return {
            'task_id': task_id,
            'status': 'pending',
            'estimated_time': 60,  # 预计 60 秒
            'poll_url': f'/api/v4/tasks/{task_id}'
        }  # < 0.1s

    async def get_task_status(self, task_id: str) -> Dict:
        """获取任务状态（轮询）"""
        task = await self.db.get_task(task_id)

        if task['status'] == 'completed':
            return {
                'task_id': task_id,
                'status': 'completed',
                'result': task['result'],
                'elapsed_time': task['elapsed_time']
            }

        elif task['status'] == 'failed':
            return {
                'task_id': task_id,
                'status': 'failed',
                'error': task['error']
            }

        else:
            return {
                'task_id': task_id,
                'status': 'running',
                'progress': task.get('progress', 0)
            }
```

## 🧠 智能路由器

### 核心：根据场景自动选择路径

```python
class SmartRouter:
    """智能路由器"""

    def __init__(self):
        self.fast_path = FastPath()
        self.normal_path = NormalPath()
        self.slow_path = SlowPath()

    async def route(self, request: Request) -> Response:
        """智能路由"""
        # 1. 分析请求类型
        request_type = self.classify_request(request)

        # 2. 选择路径
        if request_type == 'query':
            # 查询类 → Fast Path
            return await self.fast_path.handle(request)

        elif request_type == 'generation':
            # 生成类 → Normal Path
            return await self.normal_path.handle(request)

        elif request_type == 'heavy_computation':
            # 重计算 → Slow Path
            return await self.slow_path.handle(request)

        else:
            raise ValueError(f"Unknown request type: {request_type}")

    def classify_request(self, request: Request) -> str:
        """分类请求"""
        endpoint = request.url.path

        # Fast Path 端点
        fast_endpoints = [
            '/api/v4/account/topics',
            '/api/v4/account/daily-plan',
            '/api/v4/cover/candidates'
        ]

        # Normal Path 端点
        normal_endpoints = [
            '/api/v4/account/create-plan',
            '/api/v4/cover/generate',
            '/api/v4/platform/adapt'
        ]

        # Slow Path 端点
        slow_endpoints = [
            '/api/v4/causal/effect',
            '/api/v4/causal/counterfactual',
            '/api/v4/platform/compare'
        ]

        if endpoint in fast_endpoints:
            return 'query'
        elif endpoint in normal_endpoints:
            return 'generation'
        elif endpoint in slow_endpoints:
            return 'heavy_computation'
        else:
            return 'unknown'
```

## 🔄 并行执行器

### 核心：最大化并行度

```python
class ParallelExecutor:
    """并行执行器"""

    async def execute_parallel(
        self,
        tasks: List[Callable]
    ) -> List[Any]:
        """并行执行多个任务"""
        # 使用 asyncio.gather 并行执行
        results = await asyncio.gather(*tasks, return_exceptions=True)

        # 处理异常
        for i, result in enumerate(results):
            if isinstance(result, Exception):
                logger.error(f"Task {i} failed: {result}")
                results[i] = None

        return results

    async def execute_with_timeout(
        self,
        task: Callable,
        timeout: float
    ) -> Any:
        """带超时的执行"""
        try:
            result = await asyncio.wait_for(task, timeout=timeout)
            return result
        except asyncio.TimeoutError:
            logger.error(f"Task timeout after {timeout}s")
            return None

# 使用示例
executor = ParallelExecutor()

# 并行执行 3 个任务
results = await executor.execute_parallel([
    fetch_topics(),
    fetch_persona(),
    fetch_history()
])

# 带超时执行
result = await executor.execute_with_timeout(
    generate_content(),
    timeout=10.0
)
```

## 💾 多层缓存

### L1: 内存缓存（最快）

```python
class L1Cache:
    """L1 缓存：内存"""

    def __init__(self, max_size: int = 1000):
        self.cache = {}
        self.max_size = max_size
        self.access_count = {}

    def get(self, key: str) -> Optional[Any]:
        """获取缓存"""
        if key in self.cache:
            self.access_count[key] += 1
            return self.cache[key]  # < 1ms
        return None

    def set(self, key: str, value: Any):
        """设置缓存"""
        if len(self.cache) >= self.max_size:
            # LRU 淘汰
            lru_key = min(self.access_count, key=self.access_count.get)
            del self.cache[lru_key]
            del self.access_count[lru_key]

        self.cache[key] = value
        self.access_count[key] = 0
```

### L2: Redis 缓存（快）

```python
class L2Cache:
    """L2 缓存：Redis"""

    def __init__(self):
        self.redis = Redis()

    async def get(self, key: str) -> Optional[Any]:
        """获取缓存"""
        value = await self.redis.get(key)
        if value:
            return json.loads(value)  # < 10ms
        return None

    async def set(self, key: str, value: Any, ttl: int):
        """设置缓存"""
        await self.redis.setex(
            key,
            ttl,
            json.dumps(value)
        )
```

### L3: 数据库（慢）

```python
class L3Cache:
    """L3 缓存：数据库"""

    async def get(self, key: str) -> Optional[Any]:
        """从数据库获取"""
        result = await self.db.query(
            "SELECT value FROM cache WHERE key = $1",
            key
        )
        return result  # < 100ms
```

### 多层缓存管理器

```python
class MultiLevelCache:
    """多层缓存管理器"""

    def __init__(self):
        self.l1 = L1Cache()
        self.l2 = L2Cache()
        self.l3 = L3Cache()

    async def get(self, key: str) -> Optional[Any]:
        """获取缓存（L1 → L2 → L3）"""
        # 1. 尝试 L1
        value = self.l1.get(key)
        if value:
            return value  # < 1ms

        # 2. 尝试 L2
        value = await self.l2.get(key)
        if value:
            self.l1.set(key, value)  # 回填 L1
            return value  # < 10ms

        # 3. 尝试 L3
        value = await self.l3.get(key)
        if value:
            await self.l2.set(key, value, ttl=3600)  # 回填 L2
            self.l1.set(key, value)  # 回填 L1
            return value  # < 100ms

        return None

    async def set(self, key: str, value: Any, ttl: int):
        """设置缓存（写入所有层）"""
        self.l1.set(key, value)
        await self.l2.set(key, value, ttl)
        # L3 不主动写入，只作为持久化备份
```
