#### 方案 1: 三级决策权限模型

```python
class DecisionLevel(Enum):
    """决策级别"""
    FULLY_AUTO = "fully_auto"      # 完全自动
    SEMI_AUTO = "semi_auto"        # 半自动（需审批）
    MANUAL_ONLY = "manual_only"    # 仅手动

class DecisionBoundary:
    """决策边界配置"""

    # 话题选择：半自动
    topic_selection = DecisionLevel.SEMI_AUTO
    # 原因：话题敏感，需人工审核

    # 发布时间：完全自动
    publish_timing = DecisionLevel.FULLY_AUTO
    # 原因：低风险，可自动决策

    # 内容生成：半自动
    content_generation = DecisionLevel.SEMI_AUTO
    # 原因：内容质量需人工把关

    # 封面生成：完全自动
    cover_generation = DecisionLevel.FULLY_AUTO
    # 原因：可 A/B 测试，风险可控

    # 多平台发布：仅手动
    multi_platform_publish = DecisionLevel.MANUAL_ONLY
    # 原因：高风险，必须人工确认

    # 策略调整：半自动
    strategy_adjustment = DecisionLevel.SEMI_AUTO
    # 原因：影响范围大，需审批

class AutoAccountManager:
    """带决策边界的自动账号运营"""

    async def create_daily_plan(self, account_id: str):
        """创建每日计划"""
        # 1. 发现话题
        topics = await self.discover_topics()

        # 2. 检查决策级别
        if DecisionBoundary.topic_selection == DecisionLevel.SEMI_AUTO:
            # 需要人工审批
            plan = await self.create_pending_plan(topics)
            await self.notify_for_approval(plan)
            return plan

        elif DecisionBoundary.topic_selection == DecisionLevel.FULLY_AUTO:
            # 完全自动
            plan = await self.create_and_execute_plan(topics)
            return plan

        else:
            # 仅手动
            raise PermissionError("话题选择需要手动操作")
```

#### 方案 2: 风险评分与自动降级

```python
class RiskAssessment:
    """风险评估"""

    def assess_topic_risk(self, topic: str) -> float:
        """评估话题风险（0-1）"""
        risk_score = 0.0

        # 敏感词检测
        if self.contains_sensitive_words(topic):
            risk_score += 0.5

        # 争议性检测
        if self.is_controversial(topic):
            risk_score += 0.3

        # 时效性检测
        if self.is_breaking_news(topic):
            risk_score += 0.2

        return min(risk_score, 1.0)

    def auto_downgrade_decision(self, risk_score: float) -> DecisionLevel:
        """根据风险自动降级决策权限"""
        if risk_score < 0.3:
            return DecisionLevel.FULLY_AUTO
        elif risk_score < 0.7:
            return DecisionLevel.SEMI_AUTO
        else:
            return DecisionLevel.MANUAL_ONLY

# 使用示例
risk = RiskAssessment()
topic_risk = risk.assess_topic_risk("某敏感话题")

if topic_risk > 0.7:
    # 高风险话题，强制人工审核
    await notify_human_review(topic)
else:
    # 低风险话题，可自动处理
    await auto_process(topic)
```

#### 方案 3: 审计日志与回滚机制

```python
class DecisionAuditLog:
    """决策审计日志"""

    def log_decision(
        self,
        decision_type: str,
        decision_level: DecisionLevel,
        input_data: Dict,
        output_data: Dict,
        risk_score: float,
        approved_by: Optional[str] = None
    ):
        """记录决策"""
        log = {
            'timestamp': datetime.now(),
            'decision_type': decision_type,
            'decision_level': decision_level.value,
            'input': input_data,
            'output': output_data,
            'risk_score': risk_score,
            'approved_by': approved_by,
            'can_rollback': True
        }

        self.db.save(log)

    def rollback_decision(self, decision_id: str):
        """回滚决策"""
        decision = self.db.get(decision_id)

        if decision['decision_type'] == 'publish':
            # 撤回发布
            self.unpublish(decision['output']['post_id'])

        elif decision['decision_type'] == 'topic_selection':
            # 取消话题
            self.cancel_topic(decision['output']['topic_id'])
```

## ⚠️ 风险 3: 因果推断的样本规模问题

### 问题描述

PSM（Propensity Score Matching）在样本 < 1000 时非常不稳定。

**风险**: "假因果"误导决策

### 影响评估

- **决策准确性**: 🔴 低
- **系统可信度**: 🔴 低

### 缓解方案

#### 方案 1: 样本规模门槛

```python
class CausalInferenceEngine:
    """带样本规模检查的因果推断引擎"""

    MIN_SAMPLE_SIZE = 1000  # 最小样本量
    RECOMMENDED_SAMPLE_SIZE = 5000  # 推荐样本量

    async def estimate_causal_effect(
        self,
        treatment: str,
        outcome: str,
        common_causes: List[str]
    ):
        """估计因果效应"""
        # 1. 检查样本量
        sample_size = await self.get_sample_size(treatment, outcome)

        if sample_size < self.MIN_SAMPLE_SIZE:
            raise InsufficientDataError(
                f"样本量不足: {sample_size} < {self.MIN_SAMPLE_SIZE}\n"
                f"因果推断需要至少 {self.MIN_SAMPLE_SIZE} 个样本"
            )

        # 2. 计算置信度
        confidence = self.calculate_confidence(sample_size)

        # 3. 估计因果效应
        effect = await self._estimate_with_psm(treatment, outcome, common_causes)

        # 4. 添加警告
        if sample_size < self.RECOMMENDED_SAMPLE_SIZE:
            effect.warning = (
                f"样本量 {sample_size} 小于推荐值 {self.RECOMMENDED_SAMPLE_SIZE}，"
                f"结果可能不稳定"
            )

        effect.confidence = confidence
        return effect

    def calculate_confidence(self, sample_size: int) -> float:
        """计算置信度"""
        if sample_size >= self.RECOMMENDED_SAMPLE_SIZE:
            return 0.95
        elif sample_size >= self.MIN_SAMPLE_SIZE:
            # 线性插值
            ratio = (sample_size - self.MIN_SAMPLE_SIZE) / \
                    (self.RECOMMENDED_SAMPLE_SIZE - self.MIN_SAMPLE_SIZE)
            return 0.7 + 0.25 * ratio
        else:
            return 0.0
```

#### 方案 2: 降级到相关性分析

```python
class HybridCausalEngine:
    """混合因果引擎：样本不足时降级"""

    async def analyze_relationship(
        self,
        treatment: str,
        outcome: str
    ):
        """分析关系（自动选择方法）"""
        sample_size = await self.get_sample_size(treatment, outcome)

        if sample_size >= 1000:
            # 样本充足，使用因果推断
            result = await self.causal_inference(treatment, outcome)
            result.method = "causal_inference"
            result.reliability = "high"

        elif sample_size >= 100:
            # 样本中等，使用相关性分析
            result = await self.correlation_analysis(treatment, outcome)
            result.method = "correlation"
            result.reliability = "medium"
            result.warning = "样本量不足，仅显示相关性（非因果性）"

        else:
            # 样本太少，拒绝分析
            raise InsufficientDataError(
                f"样本量 {sample_size} 太少，无法进行可靠分析"
            )

        return result
```

## ⚠️ 风险 4: 多平台发布的真实复杂度

**结论**: 多平台引擎是 v4.0 里最难持续维护的部分

### 影响评估

- **维护成本**: 🔴 极高
- **失败风险**: 🔴 高
- **ROI**: 🟡 不确定

```python
class PlatformRuleEngine:
    """平台规则引擎"""

    def __init__(self):
        self.rules = self.load_rules()

    def load_rules(self) -> Dict:
        """加载平台规则（从配置文件）"""
        return {
            'xiaohongshu': {
                'content_rules': {
                    'max_title_length': 50,
                    'max_body_length': 1000,
                    'max_images': 9,
                    'forbidden_words': ['敏感词1', '敏感词2'],
                    'required_disclaimer': True
                },
                'algorithm_rules': {
                    'cold_start_duration': 3600,  # 1小时
                    'traffic_pool_levels': ['初始池', '二级池', '三级池'],
                    'key_metrics': ['完读率', '互动率', '分享率']
                },
                'compliance_rules': {
                    'need_qualification': ['医疗', '金融'],
                    'forbidden_topics': ['政治', '色情'],
                    'age_restriction': 18
                }
            }
        }

    def validate_content(self, content: UnifiedContent, platform: Platform):
        """验证内容是否符合平台规则"""
        rules = self.rules[platform.value]

        violations = []

        # 检查标题长度
        if len(content.title) > rules['content_rules']['max_title_length']:
            violations.append(f"标题超长: {len(content.title)}")

        # 检查敏感词
        for word in rules['content_rules']['forbidden_words']:
            if word in content.body:
                violations.append(f"包含敏感词: {word}")

        if violations:
            raise ContentViolationError(violations)
```

#### 方案 3: 人工审核 + 自动发布

```python
class SafeMultiPlatformEngine:
    """安全的多平台引擎：人工审核 + 自动发布"""

    async def publish_with_review(
        self,
        content: UnifiedContent,
        platforms: List[Platform]
    ):
        """带审核的发布流程"""
        # 1. 适配到各平台
        adaptations = {}
        for platform in platforms:
            adapted = await self.adapt(content, platform)
            adaptations[platform] = adapted

        # 2. 创建审核任务
        review_task = {
            'content_id': content.content_id,
            'adaptations': adaptations,
            'status': 'pending_review',
            'created_at': datetime.now()
        }

        await self.db.save(review_task)

        # 3. 通知人工审核
        await self.notify_reviewer(review_task)

        # 4. 等待审核通过后自动发布
        # （由另一个任务监听审核结果）

        return review_task['task_id']
```
