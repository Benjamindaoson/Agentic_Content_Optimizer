## 🎯 核心原则

### 1. 渐进式启用
不要一次性启用所有功能，而是分阶段启用

### 2. 人机协同
关键决策需要人工审批，而非完全自动化

### 3. 单平台优先
先做好小红书，再考虑多平台

### 4. 数据驱动
积累足够样本后再启用高级功能

### 2. 精简版自动账号运营

```python
# backend/app/growth_brain/auto_account_manager_lite.py

"""
自动化账号运营（精简版）

特点：
1. 只支持小红书
2. 关键决策需人工审批
3. 不自动发布
"""

from typing import List, Dict
from datetime import date
from enum import Enum

class ApprovalStatus(Enum):
    """审批状态"""
    PENDING = "pending"
    APPROVED = "approved"
    REJECTED = "rejected"

class AutoAccountManagerLite:
    """自动化账号运营（精简版）"""

    def __init__(self, db):
        self.db = db
        self.supported_sources = [TopicSource.XHS_TRENDING]

    async def discover_trending_topics(self, limit: int = 5) -> List[Dict]:
        """发现热点话题（仅小红书）"""
        topics = await self._fetch_xhs_trending(limit)

        # 自动过滤高风险话题
        safe_topics = []
        for topic in topics:
            risk_score = self._assess_risk(topic)
            if risk_score < 0.7:  # 只保留低中风险话题
                safe_topics.append({
                    **topic,
                    'risk_score': risk_score
                })

        return safe_topics

    async def create_daily_plan_with_approval(
        self,
        account_id: str,
        plan_date: date
    ) -> Dict:
        """创建每日计划（需审批）"""
        # 1. 发现话题
        topics = await self.discover_trending_topics(limit=5)

        # 2. 选择 Top 3
        selected_topics = topics[:3]

        # 3. 计算排程
        schedules = []
        for topic in selected_topics:
            optimal_time = await self._calculate_optimal_time(topic)
            schedules.append({
                'topic': topic,
                'scheduled_time': optimal_time,
                'status': 'pending_approval'
            })

        # 4. 创建待审批计划
        plan = {
            'plan_id': f"plan_{uuid.uuid4().hex[:16]}",
            'account_id': account_id,
            'plan_date': plan_date,
            'selected_topics': selected_topics,
            'schedules': schedules,
            'status': ApprovalStatus.PENDING,
            'created_at': datetime.now(),
            'requires_approval': True,
            'approval_url': f"/admin/plans/{plan['plan_id']}/approve"
        }

        # 5. 保存到数据库
        await self.db.save_plan(plan)

        # 6. 发送审批通知
        await self._notify_for_approval(plan)

        return plan

    def _assess_risk(self, topic: Dict) -> float:
        """评估话题风险"""
        risk_score = 0.0

        # 敏感词检测
        sensitive_words = ['政治', '色情', '暴力', '赌博']
        for word in sensitive_words:
            if word in topic['topic_name']:
                risk_score += 0.5

        # 争议性检测
        if self._is_controversial(topic):
            risk_score += 0.3

        return min(risk_score, 1.0)

    async def _notify_for_approval(self, plan: Dict):
        """发送审批通知"""
        # 发送邮件/企业微信/钉钉通知
        notification = {
            'type': 'plan_approval',
            'plan_id': plan['plan_id'],
            'message': f"新的每日计划需要审批：{len(plan['selected_topics'])} 个话题",
            'approval_url': plan['approval_url']
        }

        await self.notification_service.send(notification)

    async def approve_plan(self, plan_id: str, approved_by: str):
        """审批通过计划"""
        plan = await self.db.get_plan(plan_id)

        plan['status'] = ApprovalStatus.APPROVED
        plan['approved_by'] = approved_by
        plan['approved_at'] = datetime.now()

        await self.db.update_plan(plan)

        # 审批通过后，可以开始生成内容（但仍需人工发布）
        return plan
```

```python
# backend/app/growth_brain/multimodal_cover_engine_lite.py

"""
多模态封面引擎（精简版）

特点：
1. 只支持 DALL·E
2. 保留 A/B 测试
3. 降低成本
"""

class MultimodalCoverEngineLite:
    """多模态封面引擎（精简版）"""

    def __init__(self, db):
        self.db = db
        self.supported_generators = [CoverGenerator.DALLE]

    async def generate_cover_candidates(
        self,
        title: str,
        text: str,
        count: int = 3  # 只生成 3 个候选
    ) -> List[CoverCandidate]:
        """生成封面候选（仅 DALL·E）"""
        # 1. 创建提示词
        prompt = self._create_prompt(title, text)

        # 2. 使用 DALL·E 生成
        candidates = []
        for i in range(count):
            cover = await self._generate_with_dalle(prompt)
            candidates.append(cover)

        # 3. 预测点击率
        for candidate in candidates:
            candidate.predicted_ctr = await self._predict_ctr(candidate)

        # 4. 排序
        candidates.sort(key=lambda x: x.predicted_ctr, reverse=True)

        return candidates

    async def run_ab_test(
        self,
        candidates: List[CoverCandidate]
    ) -> str:
        """运行 A/B 测试（保留）"""
        # A/B 测试逻辑与 Full 版本相同
        test_id = await self._create_ab_test(candidates)
        return test_id
```

### 4. 精简版多平台引擎

```python
# backend/app/growth_brain/multi_platform_engine_lite.py

"""
多平台内容引擎（精简版）

特点：
1. 只支持小红书
2. 不自动发布
3. 提供发布预览
"""

class MultiPlatformEngineLite:
    """多平台内容引擎（精简版）"""

    SUPPORTED_PLATFORMS = [Platform.XIAOHONGSHU]

    async def adapt_content(
        self,
        content: UnifiedContent
    ) -> Dict:
        """适配内容（仅小红书）"""
        # 只适配小红书
        xhs_adapter = XHSAdapter()
        adapted = xhs_adapter.adapt(content)

        return {
            'platform': Platform.XIAOHONGSHU,
            'adapted_content': adapted,
            'preview_url': self._generate_preview(adapted),
            'ready_to_publish': True,
            'requires_manual_publish': True
        }

    def _generate_preview(self, adapted_content: Dict) -> str:
        """生成预览页面"""
        # 生成一个预览页面，让用户确认后手动发布
        preview_id = uuid.uuid4().hex[:16]
        preview_url = f"/admin/preview/{preview_id}"

        # 保存预览数据
        self.db.save_preview({
            'preview_id': preview_id,
            'content': adapted_content,
            'created_at': datetime.now()
        })

        return preview_url

    async def manual_publish(
        self,
        preview_id: str,
        published_by: str
    ):
        """手动发布（人工确认后）"""
        preview = await self.db.get_preview(preview_id)

        # 发布到小红书
        result = await self.xhs_publisher.publish(preview['content'])

        # 记录发布
        await self.db.record_publish({
            'preview_id': preview_id,
            'published_by': published_by,
            'published_at': datetime.now(),
            'platform_post_id': result['post_id'],
            'platform_url': result['url']
        })

        return result
```
